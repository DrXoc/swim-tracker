Google Sheets Backup Configuration

Automatically sync and backup all team attendance records, roster data, and strength logs to a Google Sheet using Google Apps Script.

Google Apps Script Snippet (Paste into Google Sheets Apps Script):
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // Sync Roster
  var rSheet = ss.getSheetByName("Roster") || ss.insertSheet("Roster");
  rSheet.clear();
  rSheet.appendRow(["ID", "First Name", "Last Name", "Grade", "Gender", "Category", "Start Date", "PIN"]);
  data.swimmers.forEach(function(s) {
    rSheet.appendRow([s.id, s.firstName, s.lastName, s.grade, s.gender, s.type, s.startDate, s.pin]);
  });

  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
